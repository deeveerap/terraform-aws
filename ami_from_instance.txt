  instance_type = "t2.micro"
  vpc_security_group_ids = [
    "sg-e55200341d67ad7f8"
  ]

  tags = {
    Name = "xfusion-ec2"
  }
}

--> Creating the AMI 
resource aws_ami_from_instance ec2ami {
  name = "xfusion-ec2-ami"
  source_instance_id = aws_instance.ec2.id
}

--> POST Creation | check the status , using the data block.

data "aws_ami" "check_status" {
  filter {
    name   = "name"
    values = ["xfusion-ec2-ami"]
  }
}

output "live_ami_state" {
  value = data.aws_ami.check_status.state
}
